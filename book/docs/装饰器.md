# 装饰器

装饰器本质是一种特殊的函数，可以在不修改原类的情况下添加额外的功能。

```typescript
function func(target: Function): void {
    console.log(target.name);
    target.func = (): string => 'Hello World!';
    target.prototype.func = (): string => 'Hello TypeScript!';
    console.log(target.prototype);
}
@func
class Person {
    constructor(
        public name: string
    ) {
        console.log(this.name);
    }
}
console.log(Person.func());
let Tom: Person = new Person('Tom');
console.log(Tom.func());
```

若装饰器函数无返回值，则被装饰的类不会被替换，若装饰器函数返回新的类，则新类将替换被装饰的类。

```typescript
function func(target: Function): object {
    return class extends Demo {
        public func2(): void {
            console.log('Hello TypeScript!');
        }
    };
}
@func
class Demo {
    public func1(): void {
        console.log('Hello World!');
    }
}
let obj: Demo = new Demo();
obj.func1();
obj.func2();
```

装饰器工厂是返回装饰器函数的函数，可以通过传入自定义参数实现更灵活的配置。

```typescript
function func(num: number): Function {
    return function (target: Function): void {
        target.prototype.func = (): void => {
            while (num > 0) {
                console.log('Hello TypeScript!');
                num--;
            }
        }
    }
}
@func(3)
class Demo {
    public func(): void {
        console.log('Hello World!');
    }
}
let obj: Demo = new Demo();
obj.func();
```
